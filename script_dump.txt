234:// File input logic (Batch)
235:document.getElementById('csvFile').addEventListener('change', (e) => {
236:    const files = e.target.files;
237:    if (!files.length) return;
238:    
239:    let filesProcessed = 0;
240:    
241:    Array.from(files).forEach(file => {
242:        Papa.parse(file, {
243:            header: true,
244:            skipEmptyLines: true,
245:            complete: (results) => {
246:                let headers = results.meta.fields || [];
247:                let detectedId = detectTableType(headers);
248:                
249:                appState[detectedId] = {
250:                    loaded: true,
251:                    headers: headers,
252:                    rows: results.data,
253:                    cellErrors: {},
254:                    autocorrectedCells: {},
255:                    hasValidated: false
256:                };
257:                
258:                let tab = document.querySelector(`.tab[data-target="${detectedId}"]`);
259:                tab.classList.remove('disabled');
260:                
261:                auditLog.push({table: detectedId, row: 'System', type: 'INFO', message: `Loaded ${results.data.length} rows.`});
262:                
263:                filesProcessed++;
264:                if (filesProcessed === files.length) {
265:                    // All files loaded. Auto-switch to table1 if loaded, or the first loaded.
266:                    if (appState['table1'].loaded) switchTab('table1');
267:                    else switchTab(detectedId);
268:                    
269:                    document.getElementById('validateBtn').disabled = false;
270:                    document.getElementById('autocorrectBtn').disabled = false;
271:                }
272:            }
273:        });
274:    });

302:function detectTableType(csvHeaders) {
303:  let joined = csvHeaders.join(',').toLowerCase();
304:  if (joined.includes('dti_no') || joined.includes('business_type')) return 'table1';
305:  if (joined.includes('business_line_code')) return 'table2';
306:  if (joined.includes('application_type')) return 'table3';
307:  if (joined.includes('application_or_no') || joined.includes('fee_code') || joined.includes('type')) return 'table4';
308:  return 'table1'; 
309:}

511:function validateSingleRow(tableId, row, rowIdx, seenLists) {
512:  let errs = {};
513:  let schema = schemas[tableId].RULES;
514:  
515:  Object.keys(schema).forEach(k => {
516:    let rule = schema[k];
517:    let val = (row[k] == null ? '' : String(row[k])).trim();
518:    
519:    if (rule.required && val === '') { errs[k] = 'Required'; return; }
520:    if (val !== '') {
521:      if (rule.min && val.length < rule.min) errs[k] = `Min ${rule.min} chars`;
522:      if (rule.max && val.length > rule.max) errs[k] = `Max ${rule.max} chars`;
523:      if (rule.allowed && !rule.allowed.includes(val)) errs[k] = `Allowed: ${rule.allowed.join(', ')}`;
524:      if (rule.numeric) {
525:        if (!/^\d+(\.\d+)?$/.test(val)) {
526:            errs[k] = 'Must be numbers only';
527:        }
528:      }
529:      let df = schemas[tableId].DATE_FIELDS || [];
530:      if (df.includes(k)) {
531:        let cleanVal = val.replace(/^'+/, '');
532:        if (!/^\d{2}\/\d{2}\/\d{4}$/.test(cleanVal)) {
533:            errs[k] = "Invalid format. Use MM/DD/YYYY";
534:        } else {
535:            let d = new Date(cleanVal);
536:            if (isNaN(d.getTime())) {
537:                errs[k] = 'Invalid Date';
538:            } else {
539:                let y = d.getFullYear();
540:                if (y < 1900 || y > 2100) errs[k] = 'Year out of range (1900-2100)';
541:            }
542:        }
543:      }
544:    }
545:  });
546:
547:  customValidateRow(tableId, row, errs, seenLists);
548:  return errs;
549:}
